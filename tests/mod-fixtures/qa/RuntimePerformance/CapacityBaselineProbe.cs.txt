        private void SelectGroundLayout(bool grouped)
        {
            if (enabled || monsters.Count != 0 || nativeGateAttempted || DolocAPI.CurrentRoom?.RoomId != TestRoom) throw new InvalidOperationException("Select once before native gate.");
            var pools = (IDictionary)AccessTools.Field(DolocAPI.EntitySystem.GetGameEntityManager<MonsterController>().GetType(), "nestedPools").GetValue(DolocAPI.EntitySystem.GetGameEntityManager<MonsterController>());
            var prefab = (GameObject)pools["fungus"].GetType().GetProperty("Prefab").GetValue(pools["fungus"]);
            var body = prefab.GetComponent<Collider2D>(); Vector2 size;
            if (body is BoxCollider2D box) size = box.size;
            else if (body is CapsuleCollider2D capsule) size = capsule.size;
            else if (body is CircleCollider2D circle) size = Vector2.one * (2 * circle.radius);
            else throw new InvalidOperationException("Unknown template body shape.");
            var scale = body.transform.lossyScale; size = Vector2.Scale(size, new Vector2(Mathf.Abs(scale.x), Mathf.Abs(scale.y)));
            if (size.x <= 0 || size.y <= 0 || Quaternion.Angle(body.transform.rotation, Quaternion.identity) > .01f) throw new InvalidOperationException("Unmeasured body transform.");
            var map = ((IMonsterHost)DolocAPI.CurrentRoom).MonsterEnv.GroundMap;
            if (settings.TreeOverview && grouped) throw new InvalidOperationException("Tree fixture is the retained separated plan.");
            var selected = new List<Vector2>(); var target = Target;
            var candidates = map.GroundPositions.Select(c => map.CellToWorldPivot(c, new Vector2(.5f, 0))).OrderBy(p => (p - target).sqrMagnitude).ThenBy(p => p.x).ThenBy(p => p.y).ToArray();
            foreach (var p in candidates)
            {
                if (!map.RaycastToGround(p + Vector2.up * .01f, out var landed) || Vector2.Distance(p, landed) > .02f || map.IsObstacleOrInvalid(map.WorldToCell(landed))) continue;
                if (selected.Any(q => RectGap(p, size, q, size) < BodyGap)) continue;
                selected.Add(p); if (selected.Count == 64) break;
            }
            groupedGround = grouped;
            var row = New("layout", "layout64"); row.kind = "fungus"; row.genType = "Ground"; row.requested = 64;
            row.candidatePositions = candidates.Length; row.validPositions = selected.Count; row.bodyShape = body.GetType().Name; row.bodyWidth = size.x; row.bodyHeight = size.y;
            row.resolvedXY = selected.SelectMany(p => new[] { p.x, p.y }).ToArray(); row.detail = "Native ground cells, nearest target first; minimum body AABB gap=0.2; no room-slot claim; " + Status();
            int requiredAnchors = grouped ? 25 : 64;
            if (selected.Count != requiredAnchors) { row.valid = false; row.error = "UnexpectedSeparatedGroundPositionCount"; }
            else if (!grouped) layout64 = selected.ToArray();
            else
            {
                groundAnchors = selected.ToArray(); anchorOccupancy = Enumerable.Range(0, 25).Select(i => i < 14 ? 3 : 2).ToArray();
                layout64 = new Vector2[64]; anchorAssignments = new int[64]; int index = 0;
                for (int a = 0; a < groundAnchors.Length; a++) for (int member = 0; member < anchorOccupancy[a]; member++)
                {
                    float dx = anchorOccupancy[a] == 3 ? (member - 1) * .45f : (member == 0 ? -.3f : .3f);
                    layout64[index] = groundAnchors[a] + new Vector2(dx, .02f); anchorAssignments[index++] = a;
                }
                AddAnchorPlan(row);
                row.requestedXY = layout64.SelectMany(p => new[] { p.x, p.y }).ToArray();
                row.detail += "; explicit grouped scenario: 14 anchors x3 + 11 x2, small input offsets; native landing and physics unchanged";
            }
            if (layout64 != null)
            {
                row.requestedXY = layout64.SelectMany(p => new[] { p.x, p.y }).ToArray();
                if (settings.TreeOverview && row.requestedXY.Where((v, i) => Mathf.Abs(v - settings.ExpectedPlanXY[i]) > .001f).Any())
                { row.valid = false; row.error = "RetainedTreePlanChanged"; layout64 = null; }
            }
            Emit(row);
        }

        private static float RectGap(Vector2 a, Vector2 sizeA, Vector2 b, Vector2 sizeB)
        {
            float x = Mathf.Abs(a.x - b.x) - (sizeA.x + sizeB.x) * .5f, y = Mathf.Abs(a.y - b.y) - (sizeA.y + sizeB.y) * .5f;
            return x <= 0 && y <= 0 ? Mathf.Max(x, y) : new Vector2(Mathf.Max(0, x), Mathf.Max(0, y)).magnitude;
        }
        private static bool Finite(float value) => !float.IsNaN(value) && !float.IsInfinity(value);
        private static Bounds CreationBounds(Collider2D body)
        {
            Vector2 size;
            if (body is BoxCollider2D box) size = box.size;
            else if (body is CapsuleCollider2D capsule) size = capsule.size;
            else if (body is CircleCollider2D circle) size = Vector2.one * (2 * circle.radius);
            else throw new InvalidOperationException("Unknown actual body shape.");
            var x = body.transform.TransformVector(new Vector3(size.x, 0, 0));
            var y = body.transform.TransformVector(new Vector3(0, size.y, 0));

                {

                }

            row.nativeLandingMatched = predictedPositions.Length == monsters.Count && row.maxLandingDeviation < .02f;
            if (groupedGround)
            {
                row.anchorPlanMatched = row.observedOccupancy.SequenceEqual(anchorOccupancy) && row.maxAnchorDistance < .75f;
                row.nativeLandingMatched = row.maxLandingDeviation < .02f;
                // The user explicitly chose 2/3 bodies per anchor. Record actual
                // density after native motion; do not force 64 non-overlapping bodies.
            }
            bool moving = kind == "mixed-ui" || kind == "code-ui";

            row.controllersMin = row.controllersMax = controllers.Count;

                layoutValid &= row.nativeLandingMatched && (groupedGround || (distinct.Count == expected && row.creationMinBodyGap >= BodyGap - .01f && row.creationOverlapPairs == 0));
            else
                layoutValid &= row.bodyBoundsSynchronized && (groupedGround || moving || (distinct.Count == expected && row.minBodyGap >= BodyGap - .01f && row.overlapPairs == 0));
            row.valid = layoutValid && row.validPositions == expected;

            if (!row.valid) row.error = "ActualPlacementFailed";
            row.detail = Status(); Emit(row); return row.valid || row.mapBoundaryObservation;
        }

        }

        private static void ReadNativeResidents(Record row)
        {
            // Window-external observations of the actual native owner, not the
            // module's own cleanup lists. Include inactive cached controllers.
            var manager = DolocAPI.EntitySystem.GetGameEntityManager<MonsterController>();
            var nested = (IDictionary)AccessTools.Field(manager.GetType(), "nestedPools").GetValue(manager);
            var sources = new SortedDictionary<string, object>(StringComparer.Ordinal) { ["<root>"] = manager.GetType().GetProperty("Pool").GetValue(manager) };
            foreach (DictionaryEntry pair in nested) sources.Add((string)pair.Key, pair.Value);
            var known = new HashSet<int>(); var seenPools = new HashSet<object>(); var observations = new List<PoolResidency>();
            foreach (var pair in sources)
            {
                if (!seenPools.Add(pair.Value)) continue; // Multiple aliases may refer to one native owner.
                var type = pair.Value.GetType();
                int[] Ids(IEnumerable items) => items.Cast<MonsterController>().Select(item => item != null ? item.GetInstanceID() : throw new InvalidOperationException("Destroyed controller in native pool.")).OrderBy(id => id).ToArray();
                var active = Ids((IEnumerable)type.GetProperty("Instances").GetValue(pair.Value));
                var cached = Ids((IEnumerable)AccessTools.Field(type, "pool").GetValue(pair.Value));
                int activeCount = (int)type.GetProperty("ActiveCount").GetValue(pair.Value), cacheCount = (int)type.GetProperty("CacheCount").GetValue(pair.Value);
                if (active.Length != activeCount || cached.Length != cacheCount) throw new InvalidOperationException("Native pool counts disagree with objects.");
                foreach (int id in active.Concat(cached)) if (!known.Add(id)) throw new InvalidOperationException("Controller has multiple native pool owners.");
                observations.Add(new PoolResidency { key = pair.Key, activeCount = activeCount, cacheCount = cacheCount, activeIds = active, cachedIds = cached });
            }
            var all = UnityEngine.Resources.FindObjectsOfTypeAll<MonsterController>().Select(c => c.GetInstanceID()).ToArray();
            if (known.Any(id => !all.Contains(id))) throw new InvalidOperationException("Native pool controller missing from Unity objects.");
            row.residentControllers = all.Length;
            row.poolKeys = observations.Select(p => p.key).ToArray();
            row.poolActiveCounts = observations.Select(p => p.activeCount).ToArray(); row.poolCacheCounts = observations.Select(p => p.cacheCount).ToArray();
            row.poolControllerIds = observations.SelectMany(p => p.activeIds.Concat(p.cachedIds)).ToArray();
            row.poolControllerOwners = observations.SelectMany((p, i) => Enumerable.Repeat(i, p.activeCount + p.cacheCount)).ToArray();
            row.poolControllerActive = observations.SelectMany(p => Enumerable.Repeat(true, p.activeCount).Concat(Enumerable.Repeat(false, p.cacheCount))).ToArray();
            // Prefabs and any objects outside the enumerated pools stay explicit.
            // Their identity set must not grow/change at the resident checkpoints.
            row.otherControllerIds = all.Where(id => !known.Contains(id)).OrderBy(id => id).ToArray();
        }

        private void StepCore()
        {
            if (transitionLabel != null)
            {
                stable = arrived && DolocAPI.CurrentRoom?.RoomId == TestRoom && !DolocAPI.gameLoop.IsPaused && !DolocAPI.gameLoop.IsGlobalPaused ? stable + 1 : 0;
                if (stable >= (settings.TreeOverview ? 120 : 3))
                {
                    SetCamera(); var row = New("entry", transitionLabel);
                    row.entryStartQpc = transitionStart; row.entryCallbackQpc = transitionCallback; row.stableReadyQpc = Stopwatch.GetTimestamp();
                    row.seconds = (transitionCallback - transitionStart) / (double)Stopwatch.Frequency;
                    row.stableReadySeconds = (row.stableReadyQpc - transitionStart) / (double)Stopwatch.Frequency;
                    transitionLabel = null; Emit(row);
                }
                else if ((Stopwatch.GetTimestamp() - transitionStart) / (double)Stopwatch.Frequency > 40) throw new InvalidOperationException("Entry timeout.");
                return;
            }
            if (window == null) return;
            if (settings.TreeOverview && (camera == null || camera.orthographicSize != 42.12f || camera.aspect != fixedCameraAspect || Vector2.Distance(camera.transform.position, new Vector2(416.25f, 216)) > .01f))
                throw new InvalidOperationException("Fixed tree camera changed.");
            if (!active)
            {
                long nowWarm = Stopwatch.GetTimestamp(); double warmFrame = (nowWarm - warmLast) * 1000.0 / Stopwatch.Frequency; warmLast = nowWarm;
                warmMaxMs = Math.Max(warmMaxMs, warmFrame);
                window.warmupSeconds = (nowWarm - warmup) / (double)Stopwatch.Frequency; window.warmupMaxFrameMs = warmMaxMs;
                if (monsters.Count == 64 && warmFrame > LongFrameStopMs)
                {
                    window.valid = false; window.error = "WarmupLongFrameStop"; window.phase = "warmup"; nativeGateFailed = true;
                    window.warmupCpuMs = CpuMs() - warmCpuStart; window.warmupPathMs = ticks.Select(t => t * 1000.0 / Stopwatch.Frequency).ToArray();
                    window.longFrameMs = warmFrame; window.longFramePathMs = ticks.Select((t, i) => (t - previousTicks[i]) * 1000.0 / Stopwatch.Frequency).ToArray();
                    var failed = window; window = null; Emit(failed); ActualPlacement(failed.label + "-warmup-stopped-positions", "warmup-stopped"); return;
                }
                Array.Copy(ticks, previousTicks, ticks.Length);
                if (window.warmupSeconds < 5) return;
                window.warmupCpuMs = CpuMs() - warmCpuStart; window.warmupPathMs = ticks.Select(t => t * 1000.0 / Stopwatch.Frequency).ToArray();
                if (monsters.Count > 0 && !ActualPlacement(window.label + "-warm-positions", "after-five-second-warmup"))
                { window.valid = false; window.error = "WarmupPlacementFailed"; nativeGateFailed = true; var failed = window; window = null; Emit(failed); return; }
                managerCount = managers.Count; window.managers = managerCount;
                bullets = managers.Values.Select(m => (List<Bullet>)AccessTools.Field(typeof(BulletManager), "bullets").GetValue(m)).ToArray();
                Array.Clear(ticks, 0, ticks.Length); frameCount = 0; gcStart = GC.CollectionCount(0); heapStart = GC.GetTotalMemory(false);
                Array.Clear(previousTicks, 0, previousTicks.Length); Array.Clear(window.calls, 0, window.calls.Length); Array.Clear(window.bytes, 0, window.bytes.Length);
                Array.Clear(window.updates, 0, window.updates.Length); Array.Clear(window.fixedUpdates, 0, window.fixedUpdates.Length); window.fired = 0;
                cpuStart = CpuMs(); firstBytes = ReadBytes(); firstAll = reader == null ? -1 : counters[2]; start = lastFrame = Stopwatch.GetTimestamp(); window.sampleStartQpc = start; active = true;
            }
            long now = Stopwatch.GetTimestamp();
            double frameMs = (now - lastFrame) * 1000.0 / Stopwatch.Frequency;
            if (now > lastFrame && frameCount < frames.Length) frames[frameCount++] = frameMs;
            if (monsters.Count == 64 && frameMs > LongFrameStopMs) { window.valid = false; window.error = "SampleLongFrameStop"; window.phase = "sample"; window.longFrameMs = frameMs; window.longFramePathMs = ticks.Select((t, i) => (t - previousTicks[i]) * 1000.0 / Stopwatch.Frequency).ToArray(); nativeGateFailed = true; }
            Array.Copy(ticks, previousTicks, ticks.Length);
            lastFrame = now;
            int count = DolocAPI.CurrentRoom.DM_monster.MonsterCount, running = 0, bulletCount = 0, ui = 0;
            foreach (var m in monsters) if (!m.IsDead && m.Controller != null && controllers.ContainsKey(m.Controller)) running++;
            foreach (var list in bullets) bulletCount += list.Count;
            foreach (var root in uiRoots) if (root != null && root.activeInHierarchy) ui++;

        private sealed class Patrol : IMonsterBehavior
        {
            private readonly ModEntry owner;
            private IMonsterBehaviorContext context;
            private MonsterActionHandle action;
            private bool hasAction, right;
            private float clock, nextAttack, nextMove, x, y;
            internal int Stops;
            internal Patrol(ModEntry owner) { this.owner = owner; }
            public void Start(IMonsterBehaviorContext value) { context = value; var self = Value(context.GetSelf()); x = self.X; y = self.Y; }
            public void Tick(float dt)
            {
                var all = owner.Before(true);
                try
                {
                    clock += dt; var token = owner.Before(true); var self = context.GetSelf(); owner.After(5, token); Value(self);
                    if (hasAction)
                    {
                        token = owner.Before(true); var status = context.GetAction(action); owner.After(6, token);
                        if (status.Success && status.Value.State == MonsterActionState.Running) return; hasAction = false;
                    }
                    if (clock >= nextAttack)
                    {
                        nextAttack = clock + 2; token = owner.Before(true); var attacks = context.GetAttacks(); owner.After(7, token);
                        var attack = attacks.Success ? attacks.Value.FirstOrDefault(a => a.Available && a.TargetInRange) : null;
                        if (attack != null) { var result = context.Attack(attack.Id, 5); if (result.Success) { action = result.Value; hasAction = true; return; } }
                    }
                    if (clock < nextMove) return; nextMove = clock + .5f;
                    var move = context.MoveTo(x + (right ? .75f : -.75f), y, 4, 20); right = !right;
                    if (move.Success) { action = move.Value; hasAction = true; }
                }
                finally { owner.After(8, all); }
            }
            public void Stop() { if (++Stops != 1) throw new InvalidOperationException("Duplicate Stop."); context = null; }
        }
